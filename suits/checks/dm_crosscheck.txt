    import io, contextlib
    from bataille import jouer_partie

    def play_to_cycle(p1, p2, limit=10 ** 7):
        with contextlib.redirect_stdout(io.StringIO()):
            r = jouer_partie(list(p1), list(p2), 'naturelle', 'naturelle', limite_cartes_posees=limit)
        return list(r[2]) if r[0] == 0 and r[1] == 0 and isinstance(r[2], tuple) else None
